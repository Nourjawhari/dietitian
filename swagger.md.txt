openapi: 3.0.3

info:
  title: Dietitian Booking System API
  version: 1.1.0
  description: |
    API for the dietitian website, patient booking, health records, and administration.

    Conventions:
    - All IDs are opaque strings (`varchar` in the database).
    - Dates use `YYYY-MM-DD`; times use 24-hour `HH:mm` in the clinic's local timezone.
    - Timestamps are ISO 8601 date-times in UTC.
    - List endpoints are paginated with `page` / `page_size` and return a `pagination` object.
    - Errors always return the `Error` schema.

servers:
  - url: /api

tags:
  - name: Public
    description: Unauthenticated website content and slot availability (bilingual en/ar).
  - name: Authentication
    description: Patient registration, login and logout for patients and the administrator (JWT bearer tokens).
  - name: Patients
    description: The signed-in patient's own profile and consent settings.
  - name: Appointments
    description: Booking, cancelling, rescheduling and listing the signed-in patient's own appointments, including history.
  - name: Health Records
    description: Patient health intake (self-service) and dietitian-managed health records.
  - name: Body Composition
    description: Upload of body composition reports, AI-based metric extraction, and dietitian review/confirmation.
  - name: Admin
    description: Administrator-only management of patients, appointments, appointment slots and appointment settings.
  - name: Pages
    description: Administrator management of website pages and their sections (content editable without code changes).
  - name: Health Tips
    description: Health tips shown on the website — public read and administrator management.
  - name: Contact
    description: Public consultation/contact form and administrator handling of submissions.
  - name: Notifications
    description: Administrator view of appointment notifications (confirmations and reminders via WhatsApp with email fallback).
  - name: Reports
    description: Administrator export of appointment and patient reports as CSV or PDF.
  - name: Statistics
    description: Administrator dashboard figures — booked clients, chronic-condition and obesity counts/percentages.
